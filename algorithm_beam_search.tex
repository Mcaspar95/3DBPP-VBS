\begin{algorithm}
\caption{Beam Search for the MHKP}
\label{alg:bs3d}
\small
\setlength{\baselineskip}{10pt}
\begin{algorithmic}[1]

\Procedure{BeamSearch}{$w, m, T^{\max}$}

    \State $N_0 \gets ((0,0,0,W_j,H_j,L_j), \emptyset, I)$ \Comment{Full space and no items packed}
    \State $(P^\star, f^\star) \gets \text{FCH}(N_0)$
      %\Comment{Greedy rollout, see Algorithm \ref{alg:fch3d}}
    \State $\mathcal{W} \gets \{N_0\}$
      \Comment{Beam of the current level}

    \While{$\mathcal{W} \neq \varnothing$ \textbf{and} elapsed time $\le T^{\max}$}

        \State $\mathcal{C} \gets \varnothing$
        \Comment{Candidates of every node in $\mathcal{W}$}
        \ForAll{$N \in \mathcal{W}$}
            \ForAll{$(j,s,i,o) \in$ \Call{Expand}{$N, m$}} 
                \State $N' \gets$ $N$
                \State Place item $i$ in space $s$ of bin $j$, rotation $o$
                \State Update $N'$
                \State $(P', f') \gets \text{FCH}(N')$
                  \Comment{Score the node}
                \If{$f' > f^\star$}
                    \State $(P^\star, f^\star) \gets (P', f')$
                      \Comment{Better solution is found}
                \EndIf
                \State $\mathcal{C} \gets \mathcal{C} \cup \{(N', f')\}$
            \EndFor
        \EndFor

        \State $\mathcal{W} \gets$ the $w$ nodes of $\mathcal{C}$ with largest $f'$
          \Comment{Prune to the beam width}

    \EndWhile

    \State \textbf{return} $(P^\star, f^\star)$

\EndProcedure

\Statex

\Procedure{Expand}{$N, m$}

    \ForAll{bin $j$ by non-increasing priority $p_j$}
        \ForAll{space $s \in \mathcal{S}_j$ in DBL order}
        \State $\mathcal{F} \gets$ unplaced items of $N$ that fit in $s$ in some $o$
            \If{$\mathcal{F} \neq \varnothing$}
                \State $\mathcal{F}' \gets \varnothing$,
                       \quad $k \gets \lfloor m/|O| \rfloor$
                  \Comment{Sample $k$ items}
                \While{$|\mathcal{F}'| < k$ \textbf{and}
                       $\mathcal{F} \setminus \mathcal{F}' \neq \varnothing$}
                    \State Draw $i \in \mathcal{F} \setminus \mathcal{F}'$ with
                           probability $\dfrac{v_i}
                           {\sum_{i' \in \mathcal{F} \setminus \mathcal{F}'} v_{i'}}$
                    \State $\mathcal{F}' \gets \mathcal{F}' \cup \{i\}$
                \EndWhile
                \State \textbf{return} $\{(j,s,i,o) : i \in \mathcal{F}',\, o \in O \text{ feasible}\}$
            \EndIf
        \EndFor
    \EndFor

    \State \textbf{return} $\varnothing$ \Comment{No bin has a space that admits any remaining item}

\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{Fast constructive heuristic (\textsc{FCH})}
\label{alg:fch3d}
\small
\setlength{\baselineskip}{10pt}
\begin{algorithmic}[1]

\Procedure{FCH}{$N$}
    \Comment{Greedily complete $N$}

    \State $N' \gets$ $N$
    \State $\mathcal{B}' \gets$ items not yet placed in $N'$

    \ForAll{bin $j$ by non-increasing priority $p_j$}
      \Comment{Small bins first}

        \While{\textsc{true}}
            \State $\mathcal{S}'_j \gets$ spaces of bin $j$, sorted by DBL
              \Comment{Lowest $y$, then $z$, then $x$}
            \State $(s^\ast, i^\ast, o^\ast) \gets \varnothing$
              \Comment{Best placement found so far}

            \ForAll{$s \in \mathcal{S}'_j$ \textbf{while} $s^\ast = \varnothing$}
                \ForAll{$i \in \mathcal{B}'$ by non-increasing volume
                        \textbf{while} $s^\ast = \varnothing$}
                    \ForAll{$o \in O$ \textbf{while} $s^\ast = \varnothing$}
                        \If{\textsc{Fits}$(N', j, s, i, o)$}
                            \State $(s^\ast, i^\ast, o^\ast) \gets (s, i, o)$
                        \EndIf
                    \EndFor
                \EndFor
            \EndFor

            \If{$s^\ast = \varnothing$}
                \State \textbf{break} \Comment{Bin $j$ is closed}
            \EndIf

            \State Place item $i^\ast$ at the DBL corner of $s^\ast$
                   in rotation $o^\ast$
            \State $\mathcal{B}' \gets \mathcal{B}' \setminus \{i^\ast\}$
            \State $\mathcal{S}_j \gets$ \Call{UpdateFreeSpaces}{$\mathcal{S}'_j, i^\ast$}
        \EndWhile

    \EndFor

    \State $f \gets \dfrac{\sum_{j \in J} p_j\, v_j(N')}
                          {\sum_{j \in J} p_j\, V_j}$
      \Comment{Objective value}

    \State \textbf{return} $(N', f)$

\EndProcedure

\Statex

\Procedure{UpdateFreeSpaces}{$\mathcal{S}_j, i$}
    \Comment{See Parre\~{n}o et al. (2008)}

    \State $\mathcal{S}'' \gets \varnothing$ \Comment{New maximal free space of bin $j$}

    \ForAll{$s \in \mathcal{S}_j$}
        \If{$i$ does not intersect $s$}
            \State $\mathcal{S}'' \gets \mathcal{S}'' \cup \{s\}$
              \Comment{Untouched space is kept as is}
        \Else
            \ForAll{axis $a \in \{x,y,z\}$}
                \ForAll{side $d$ of $i$ along $a$ that lies inside $s$}
                    \State $\mathrm{slab} \gets$ the part of $s$ on side $d$
                           of $i$ along axis $a$
                    \State $\mathcal{S}'' \gets \mathcal{S}'' \cup \{\mathrm{slab}\}$
                \EndFor
            \EndFor
              \Comment{Up to six slabs per $s$}
        \EndIf
    \EndFor

    \ForAll{$s \in \mathcal{S}''$}
        \If{$s$ is contained in some other $s' \in \mathcal{S}''$}
            \State $\mathcal{S}'' \gets \mathcal{S}'' \setminus \{s\}$
              \Comment{Discard $s$ so the list stays bounded}
        \EndIf
    \EndFor

    \State \textbf{return} $\mathcal{S}''$

\EndProcedure
\end{algorithmic}
\end{algorithm}

\providecommand{\up}{\rule{0pt}{2.6ex}}
\providecommand{\down}{\rule[-1.2ex]{0pt}{0pt}}
\providecommand{\TABLE}[3]{\caption{#1}#2#3}

\begin{table}[htbp]
\TABLE
{Common notation for beam search \label{tab:BS}}
{\begin{tabular}{@{}l@{\quad}c@{}}
\hline\up
Symbol              & Explanation            \\ \hline\up
$T^{max}$ & Time limit in seconds               \\
$w$ & Beam width (nodes retained at each level of the search tree)               \\
$m$ & Maximum number of child nodes generated per expanded node               \\
$O$ & Set of allowed rotations of an item               \\
$I$ & Set of all items of the instance               \\
$\kappa$ & Bins sorted by non-increasing priority $p_j$               \\
$p_j$ & Priority of bin $j$, $p_j = 1/V_j$ (small bins first)               \\
$V_j$ & Volume of bin $j$               \\
$(W_j, H_j, L_j)$ & Width, height and length of bin $j$               \\
$s$ & Maximal free space $(x,y,z,W,H,L)$: its DBL corner and its extents               \\
$v_j(N)$ & Volume packed into bin $j$ in node $N$               \\
$N$ & Node, a partial packing $(\mathcal{S}, P, \mathcal{B})$               \\
$\mathcal{S}_j$ & Set of maximal free spaces of bin $j$               \\
$\mathcal{B}$ & Set of items not yet placed               \\
$\mathcal{F}$ & Set of unplaced items that fit the current space $s$ in some rotation               \\
$\mathcal{F}'$ & Volume-weighted sample of $k = \lfloor m/|O| \rfloor$ items drawn from $\mathcal{F}$               \\
$v_i$ & Volume of item $i$, the sampling weight within $\mathcal{F}$               \\
$\mathcal{W}$ & Beam, the set of nodes retained at the current level               \\
$\mathcal{C}$ & Set of candidate child nodes generated at the current level               \\
$\mathcal{A}$ & Set of available actions $(j,s,i,o)$ returned by \textsc{Expand}               \\
$a$ & Action $(j,s,i,o)$: place item $i$ in space $s$ of bin $j$ in rotation $o$               \\
$g(N)$ & Evaluation function scoring node $N$, obtained by the greedy rollout \textsc{FCH}               \\
$f$ & Objective value of a complete solution, $f \in [0,1]$               \\
$(P^\star, f^\star)$ & Best complete solution found and its objective value  \down\\ \hline
\end{tabular}}{}
\end{table}
